/**
 * Base URL of the FastAPI backend. Falls back to localhost:8000 for local
 * dev if NEXT_PUBLIC_API_URL isn't set (see .env.local.example).
 */
export const API_BASE_URL =
  process.env.NEXT_PUBLIC_API_URL?.replace(/\/+$/, "") || "http://localhost:8000";

/** Thrown by apiFetch for both network failures and non-2xx responses. */
export class ApiError extends Error {
  status?: number;
  constructor(message: string, status?: number) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

const DEFAULT_TIMEOUT_MS = 30_000;

async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), DEFAULT_TIMEOUT_MS);

  let res: Response;
  try {
    res = await fetch(`${API_BASE_URL}${path}`, {
      ...init,
      signal: controller.signal,
      headers: {
        "Content-Type": "application/json",
        Accept: "application/json",
        ...init?.headers,
      },
    });
  } catch (err) {
    if (err instanceof DOMException && err.name === "AbortError") {
      throw new ApiError("انتهت مهلة الاتصال بالخادم. حاول مرة أخرى.");
    }
    // fetch() rejects with a TypeError for connection-refused / DNS / CORS
    // failures -- i.e. exactly the "backend isn't running" case.
    throw new ApiError(
      "تعذر الاتصال بالخادم. تأكد من أن الخادم يعمل ثم حاول مرة أخرى."
    );
  } finally {
    clearTimeout(timeout);
  }

  if (!res.ok) {
    let message = `حدث خطأ في الخادم (رمز ${res.status}).`;
    try {
      const body = (await res.json()) as { message?: string; detail?: string };
      message = body.message || body.detail || message;
    } catch {
      // response body wasn't JSON -- keep the generic message
    }
    throw new ApiError(message, res.status);
  }

  try {
    return (await res.json()) as T;
  } catch {
    throw new ApiError("استجابة غير صالحة من الخادم.");
  }
}

"use client";

import { useCallback, useRef, useState } from "react";
import { postChat, ApiError } from "@/lib/api";
import type { ChatTurn } from "@/types";

function newId(): string {
  if (typeof crypto !== "undefined" && "randomUUID" in crypto) {
    return crypto.randomUUID();
  }
  return `${Date.now()}-${Math.random().toString(36).slice(2)}`;
}

interface UseChatResult {
  turns: ChatTurn[];
  loading: boolean;
  ask: (question: string) => Promise<void>;
  reset: () => void;
}

/** Manages the question/answer history for the /ask page. */
export function useChat(): UseChatResult {
  const [turns, setTurns] = useState<ChatTurn[]>([]);
  const [loading, setLoading] = useState(false);
  // Keep conversationId in a ref so it persists across renders without
  // causing extra re-renders and is always up-to-date inside the closure.
  const conversationIdRef = useRef<string | undefined>(undefined);

  const ask = useCallback(async (question: string) => {
    const trimmed = question.trim();
    if (!trimmed || loading) return;

    const id = newId();
    const createdAt = new Date().toISOString();

    setTurns((prev) => [...prev, { id, question: trimmed, createdAt }]);
    setLoading(true);

    try {
      const response = await postChat({
        question: trimmed,
        conversationId: conversationIdRef.current,
      });

      // Save the conversation ID returned by the backend so follow-up
      // questions are added to the same conversation thread.
      conversationIdRef.current = response.conversationId;

      setTurns((prev) =>
        prev.map((t) => (t.id === id ? { ...t, answer: response.answer } : t))
      );
    } catch (err) {
      const message =
        err instanceof ApiError
          ? err.message
          : "حدث خطأ غير متوقع. حاول مرة أخرى.";
      setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, error: message } : t)));
    } finally {
      setLoading(false);
    }
  }, [loading]);

  const reset = useCallback(() => {
    setTurns([]);
    conversationIdRef.current = undefined;
  }, []);

  return { turns, loading, ask, reset };
}

def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(_bearer_scheme),
    db: Session = Depends(get_db),
) -> User:
    if credentials is None:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Not authenticated")
    payload = decode_access_token(credentials.credentials)
    if not payload or "sub" not in payload:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid or expired token")
    user = db.query(User).filter(User.id == payload["sub"]).first()
    if not user or not user.is_active:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="User not found or inactive")
    return user

    if user.role != "admin":
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Admin access required")
    return user

class GeminiUnavailableError(Exception):
    """Raised when the Gemini API cannot be reached or is not configured."""


def _get_client():
    """Lazily construct a genai client so import doesn't fail without a key."""
    if not settings.GEMINI_API_KEY:
        raise GeminiUnavailableError("GEMINI_API_KEY is not configured.")
    from google import genai

    return genai.Client(api_key=settings.GEMINI_API_KEY)


def call_gemini_with_retry(
    contents: str,
    system_instruction: str,
    response_mime_type: str = "application/json",
    max_output_tokens: int = 8192,
    model: Optional[str] = None,
) -> str:
    """
    Calls Gemini via google-genai, retrying once after a short sleep if the
    API responds with 429 (rate limited) or 503 (overloaded).

    Returns the raw text of the model's response. Raises GeminiUnavailableError
    if the API key isn't configured or both attempts fail.
    """
    from google.genai import types
    from google.genai import errors as genai_errors

    client = _get_client()
    model_name = model or settings.GEMINI_MODEL

    attempts = 0
    last_error: Optional[Exception] = None
    while attempts < 2:
        attempts += 1
        try:
            response = client.models.generate_content(
                model=model_name,
                contents=contents,
                config=types.GenerateContentConfig(
                    system_instruction=system_instruction,
                    response_mime_type=response_mime_type,
                    max_output_tokens=max_output_tokens,
                ),
            )
            return response.text or ""
        except genai_errors.APIError as exc:  # type: ignore[attr-defined]
            last_error = exc
            code = getattr(exc, "code", None)
            if code in (429, 503) and attempts < 2:
                logger.warning("Gemini APIError %s, retrying once after backoff...", code)
                time.sleep(2)
                continue
            raise GeminiUnavailableError(f"Gemini API call failed: {exc}") from exc
        except Exception as exc:  # pragma: no cover - defensive
            last_error = exc
            raise GeminiUnavailableError(f"Gemini API call failed: {exc}") from exc

    raise GeminiUnavailableError(f"Gemini API call failed after retries: {last_error}")

"""
Detect explicit legal-article references in Arabic text, e.g. "المادة 341"
or "المادة رقم 75 من قانون العقوبات", so exact-lookup can be prioritized
over fuzzy keyword search.
"""
import re
from dataclasses import dataclass
from typing import List, Optional

from app.utils.arabic import normalize_arabic_digits

# Matches: المادة (رقم)? <number> (من <law name>)?
_ARTICLE_REF_RE = re.compile(
    r"(?:المادة|مادة)\s*(?:رقم)?\s*\(?\s*(\d+)\s*\)?"
    r"(?:\s*(?:مكرر|مكررا|مكرراً))?"
    r"(?:\s*من\s+([؀-ۿ\s]{3,60}?))?(?=[.,؛،؟!]|$)"
)


@dataclass
class ArticleReference:
    article_number: str
    law_name: Optional[str] = None
    raw_match: str = ""


def extract_article_references(text: str) -> List[ArticleReference]:
    """Find all 'المادة N (من قانون كذا)' style references in free text."""
    if not text:
        return []
    normalized = normalize_arabic_digits(text)
    refs: List[ArticleReference] = []
    for match in _ARTICLE_REF_RE.finditer(normalized):
        number = match.group(1)
        law_name = match.group(2).strip() if match.group(2) else None
        refs.append(ArticleReference(article_number=number, law_name=law_name, raw_match=match.group(0)))
    return refs


def has_exact_reference(text: str) -> bool:
    return len(extract_article_references(text)) > 0

"""
Merge exact-reference, keyword, and vector search into one ranked result list.
Exact article references always win and are placed first.
Now accepts an optional `legal_area` parameter to pre-filter results by law category.
"""
from typing import List, Optional

from sqlalchemy.orm import Session

from app.models.legal import Article, Law

from app.search.keyword import keyword_search_articles, keyword_search_judgments
from app.search.vector import vector_search

_SNIPPET_SIZE = 500  # characters, keep consistent with keyword.py


def hybrid_search(
    db: Session,
    query: str,
    limit: int = 10,
    legal_area: Optional[str] = None,
) -> List[SearchResultItem]:
    """
    Hybrid search merging:
      1. Exact article-number references (score=10.0, highest priority)
      2. Keyword search (articles + judgments, scored by match density)
      3. Vector/semantic search (cosine similarity, if embeddings are available)

    When `legal_area` is provided (e.g. "قانون جنائي"), results are filtered to
    laws whose category matches -- reducing irrelevant cross-domain noise.
    """
    results: List[SearchResultItem] = []
    seen_ids = set()

    # Build a set of law_ids matching the legal_area for filtering.
    area_law_ids: Optional[set] = None
    if legal_area and legal_area.strip() and legal_area.strip() != "غير محدد":
        area_term = legal_area.strip()
        matching_laws = (
            db.query(Law.id)
            .filter(Law.category.ilike(f"%{area_term}%"))
            .all()
        )
        if matching_laws:
            area_law_ids = {row[0] for row in matching_laws}
            # If the filter matches no laws at all, don't restrict (avoids empty results).

    # 1. Exact article-number references take priority.
    for ref in extract_article_references(query):
        q = db.query(Article).filter(Article.article_number == ref.article_number)
        if area_law_ids:
            q = q.filter(Article.law_id.in_(area_law_ids))
        rows = q.limit(5).all()
        for a in rows:
            if a.id in seen_ids:
                continue
            seen_ids.add(a.id)
            results.append(
                SearchResultItem(
                    result_type="article",
                    id=a.id,
                    title=f"المادة {a.article_number}",
                    snippet=(a.text or "")[:_SNIPPET_SIZE],
                    score=10.0,
                )
            )

    # 2. Keyword search (articles + judgments).
    for item in keyword_search_articles(db, query, limit=limit):
        if item.id not in seen_ids:
            seen_ids.add(item.id)
            results.append(item)
    for item in keyword_search_judgments(db, query, limit=limit):
        if item.id not in seen_ids:
            seen_ids.add(item.id)
            results.append(item)

    # 3. Vector search (cosine similarity from JSONL embeddings if available).
    for item in vector_search(db, query, limit=limit):
        if item.id not in seen_ids:
            seen_ids.add(item.id)
            results.append(item)

    results.sort(key=lambda r: r.score, reverse=True)
    return results[:limit]

def _match_score(text: str, term: str) -> float:
    """Score based on term frequency relative to text length (avoids flat 1.0)."""
    if not text or not term:
        return 0.5
    text_lower = text.lower()
    term_lower = term.lower()
    count = text_lower.count(term_lower)
    if count == 0:
        return 0.5
    # Normalised TF: more occurrences relative to text length = higher score.
    tf = count / max(len(text.split()), 1)
    # Cap between 0.5 and 3.0 so keyword hits stay below exact-ref score of 10.0.
    return min(3.0, 0.5 + tf * 20)

"""
Semantic/vector search using cosine similarity.

Implementation loads embeddings from data/embeddings/articles.jsonl (generated
by scripts/create_embeddings.py) into memory on first call, then embeds the
query via Gemini text-embedding-004 and ranks by cosine similarity.

Falls back to an empty result set when:
  - The JSONL file doesn't exist yet (no embeddings generated)
  - The Gemini API key isn't configured
  - Any unexpected error occurs

This keeps the hybrid search callers safe regardless of state.

TODO: once on Postgres + pgvector, replace the in-memory approach with:
    db.execute("SELECT id FROM embeddings ORDER BY embedding <-> :q LIMIT :k", ...)
"""
import json
import logging
import math
from pathlib import Path
from typing import List, Optional

from sqlalchemy.orm import Session

from app.schemas.legal import SearchResultItem

logger = logging.getLogger(__name__)

_EMBEDDINGS_JSONL = Path("data/embeddings/articles.jsonl")
_SIMILARITY_THRESHOLD = 0.50   # minimum cosine similarity to include a result
_SCORE_SCALE = 5.0             # multiply cosine [0,1] to sit below exact-ref score of 10

# Module-level cache so we only read the file once per process.
_CACHE: Optional[List[dict]] = None


def _load_embeddings() -> List[dict]:
    global _CACHE
    if _CACHE is not None:
        return _CACHE
    if not _EMBEDDINGS_JSONL.exists():
        logger.debug("vector_search: %s not found -- returning empty results", _EMBEDDINGS_JSONL)
        _CACHE = []
        return _CACHE
    data: List[dict] = []
    try:
        with open(_EMBEDDINGS_JSONL, encoding="utf-8") as fh:
            for line in fh:
                line = line.strip()
                if line:
                    obj = json.loads(line)
                    if obj.get("embedding"):
                        data.append(obj)
        logger.info("vector_search: loaded %d embeddings from %s", len(data), _EMBEDDINGS_JSONL)
    except Exception as exc:
        logger.warning("vector_search: failed to load embeddings: %s", exc)
        data = []
    _CACHE = data
    return _CACHE


def _cosine_similarity(a: List[float], b: List[float]) -> float:
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(x * x for x in b))
    if norm_a == 0.0 or norm_b == 0.0:
        return 0.0
    return dot / (norm_a * norm_b)


def _embed_query(query: str) -> Optional[List[float]]:
    """Embed a query string with Gemini text-embedding-004, task type RETRIEVAL_QUERY."""
    try:
        from app.config import settings
        if not settings.GEMINI_API_KEY:
            return None
        from google import genai
        from google.genai import types as genai_types

        client = genai.Client(api_key=settings.GEMINI_API_KEY)
        result = client.models.embed_content(
            model="text-embedding-004",
            contents=query,
            config=genai_types.EmbedContentConfig(task_type="RETRIEVAL_QUERY"),
        )
        return list(result.embeddings[0].values)
    except Exception as exc:
        logger.warning("vector_search: could not embed query: %s", exc)
        return None


def vector_search(db: Session, query: str, limit: int = 10) -> List[SearchResultItem]:
    """
    Returns up to `limit` results ranked by semantic similarity.
    Returns an empty list if embeddings aren't available or the API call fails.
    """
    embeddings = _load_embeddings()
    if not embeddings:
        return []

    query_vec = _embed_query(query)
    if query_vec is None:
        return []

    scored: List[tuple] = []
    for item in embeddings:
        emb = item.get("embedding", [])
        if not emb:
            continue
        sim = _cosine_similarity(query_vec, emb)
        if sim >= _SIMILARITY_THRESHOLD:
            scored.append((sim, item))

    scored.sort(key=lambda x: x[0], reverse=True)

    results: List[SearchResultItem] = []
    for sim, item in scored[:limit]:
        results.append(
            SearchResultItem(
                result_type="article",
                id=item.get("article_id", ""),
                title=f"المادة {item.get('article_number', '')}",
                snippet=(item.get("text_preview") or "")[:500],
                score=float(sim) * _SCORE_SCALE,
            )
        )
    return results

class InMemoryRateLimiter:
    def __init__(self, requests_per_minute: int = 30):
        self.requests_per_minute = requests_per_minute
        self._hits: Dict[str, List[float]] = defaultdict(list)
        self._lock = Lock()

    def is_allowed(self, key: str) -> bool:
        now = time.time()
        window_start = now - 60
        with self._lock:
            hits = [t for t in self._hits[key] if t > window_start]
            if len(hits) >= self.requests_per_minute:
                self._hits[key] = hits
                return False
            hits.append(now)
            self._hits[key] = hits
            return True

from datetime import datetime, timezone
from typing import Optional

from sqlalchemy.orm import Session

from app.models.users import User
from app.repositories.user_repository import UserRepository
from app.security.hashing import hash_password, verify_password
from app.security.jwt import create_access_token


class AuthError(Exception):
    """Raised for user-facing auth failures (bad credentials, duplicate email...)."""


class AuthService:
    def __init__(self, db: Session):
        self.db = db
        self.users = UserRepository(db)

    def register(self, email: str, password: str, full_name: Optional[str] = None) -> User:
        existing = self.users.get_by_email(email)
        if existing:
            raise AuthError("Email already registered.")
        user = User(email=email, hashed_password=hash_password(password), full_name=full_name)
        return self.users.create(user)

    def authenticate(self, email: str, password: str) -> User:
        user = self.users.get_by_email(email)
        if not user or not verify_password(password, user.hashed_password):
            raise AuthError("Invalid email or password.")
        if not user.is_active:
            raise AuthError("This account is disabled.")
        user.last_login_at = datetime.now(timezone.utc)
        self.db.commit()
        return user

    def issue_token(self, user: User) -> str:
        return create_access_token(subject=user.id, role=user.role)

"""
Orchestrates the chat pipeline: analyzer -> hybrid search -> RAG engine ->
answer_builder, then persists the conversation/message/citations.
"""
import json
import logging
from typing import Any, Dict, List, Optional, Tuple

from sqlalchemy.orm import Session

from app.models.system import AnswerCitation
from app.models.users import Conversation, Message
from app.rag import answer_builder
from app.rag.analyzer import analyze_question
from app.rag.engine import GeminiUnavailableError, generate_answer
from app.repositories.user_repository import ConversationRepository
from app.schemas.legal import SearchResultItem
from app.search.hybrid import hybrid_search

logger = logging.getLogger(__name__)

# Maximum number of previous conversation turns to include as context.
_HISTORY_TURNS = 4


class ChatService:
    def __init__(self, db: Session):
        self.db = db
        self.conversations = ConversationRepository(db)

    def _get_or_create_conversation(
        self, user_id: str, conversation_id: Optional[str], first_message: str
    ) -> Conversation:
        if conversation_id:
            conv = self.conversations.get(conversation_id)
            if conv and conv.user_id == user_id:
                return conv
        title = (first_message[:80] + "...") if len(first_message) > 80 else first_message
        return self.conversations.create(user_id=user_id, title=title)

    def _get_history(self, conversation_id: str) -> List[Dict[str, str]]:
        """Fetch the last N user+assistant messages for context."""
        try:
            messages = (
                self.db.query(Message)
                .filter(Message.conversation_id == conversation_id)
                .order_by(Message.created_at.desc())
                .limit(_HISTORY_TURNS * 2)
                .all()
            )
            # Reverse so they're in chronological order.
            return [{"role": m.role, "content": m.content} for m in reversed(messages)]
        except Exception as exc:
            logger.warning("Could not fetch conversation history: %s", exc)
            return []

    def ask(
        self,
        user_id: str,
        question: str,
        conversation_id: Optional[str] = None,
    ) -> Tuple[Conversation, Message, List[SearchResultItem], Dict[str, Any]]:
        """
        Returns (conversation, assistant_message, search_results, structured_answer_dict).
        The structured_answer_dict matches LegalAnswerOut and is ready for the API response.
        """
        conversation = self._get_or_create_conversation(user_id, conversation_id, question)
        self.conversations.add_message(conversation.id, role="user", content=question)

        # 1. Fetch conversation history for context.
        history = self._get_history(conversation.id)

        # 2. Analyzer: understand the question and extract legal_area.
        analysis = analyze_question(question)
        legal_area = analysis.get("legal_area") or None

        # 3. Search: hybrid retrieval over the legal corpus, filtered by legal area.
        search_results = hybrid_search(self.db, question, limit=8, legal_area=legal_area)

        # 4. RAG engine: call Gemini with grounded context, or fall back.
        try:
            raw = generate_answer(question, search_results, analysis, history)
            answer_dict = answer_builder.parse_gemini_answer(raw)
        except GeminiUnavailableError as exc:
            logger.warning("Chat: Gemini unavailable, using fallback answer: %s", exc)
            answer_dict = answer_builder.build_fallback_answer(
                question=question,
                analysis=analysis,
                context_results=search_results,
                note="خدمة الذكاء الاصطناعي (Gemini) غير مُهيأة حالياً على هذا الخادم.",
            )

        # Store structured answer as JSON string in the DB for history retrieval.
        answer_json = json.dumps(answer_dict, ensure_ascii=False)
        assistant_message = self.conversations.add_message(
            conversation.id, role="assistant", content=answer_json
        )

        # 5. Persist citations for traceability (answer_citations table).
        for result in search_results[:5]:
            citation = AnswerCitation(
                message_id=assistant_message.id,
                source_type=result.result_type,
                article_id=result.id if result.result_type == "article" else None,
                judgment_id=result.id if result.result_type == "judgment" else None,
                citation_text=result.title,
                relevance_score=result.score,
            )
            self.db.add(citation)
        self.db.commit()

        return conversation, assistant_message, search_results, answer_dict

"""Small Arabic text helpers used across search/ingestion/rag."""
import re

_ARABIC_DIACRITICS = re.compile(r"[ً-ٰٟۖ-ۭ]")
_ARABIC_INDIC_DIGITS = str.maketrans("٠١٢٣٤٥٦٧٨٩", "0123456789")


def strip_diacritics(text: str) -> str:
    """Remove Arabic diacritical marks (tashkeel) for more robust matching."""
    return _ARABIC_DIACRITICS.sub("", text or "")


def normalize_arabic_digits(text: str) -> str:
    """Convert Arabic-Indic digits (٠-٩) to ASCII digits (0-9)."""
    return (text or "").translate(_ARABIC_INDIC_DIGITS)


def normalize_alef_forms(text: str) -> str:
    """Collapse alef variants (أ إ آ) to bare alef (ا) for lenient matching."""
    if not text:
        return text
    for ch in ("أ", "إ", "آ"):
        text = text.replace(ch, "ا")
    return text


def simple_normalize(text: str) -> str:
    """Light normalization pipeline used by keyword search."""
    text = strip_diacritics(text)
    text = normalize_arabic_digits(text)
    text = normalize_alef_forms(text)
    text = re.sub(r"\s+", " ", text).strip()
    return text

def clamp_pagination(skip: int = 0, limit: int = 20, max_limit: int = 100) -> Tuple[int, int]:
